1. What Is a Parameterized DAG?

A parameterized DAG is a reusable Airflow workflow whose runtime behavior changes according to values such as table name, file name, date, customer, region or environment.

One DAG definition + different runtime values = reusable business runs
Restaurant = DAG. Kitchen steps = Tasks. Food selection = Parameter.

2. Why Use It?

Less code

Replace duplicate DAG files.

Maintenance

Fix workflow logic once.

Reusable

Process tables, dates and customers.

Flexible

Change values when triggering.

Validated

Use type, enum and range rules.

Auditable

Run configuration is recorded.

9. Complete Real-World DAG

Open complete executable code
from __future__ import annotations

import csv
import json
import os
import re
import shutil
from pathlib import Path
from typing import Any

import pendulum

from airflow.sdk import DAG, Param
from airflow.providers.standard.operators.empty import EmptyOperator
from airflow.providers.standard.operators.python import PythonOperator


LAB_HOME = Path(
    os.environ.get(
        "AIRFLOW_PARAMETER_LAB_HOME",
        str(Path.home() / "airflow_parameterized_lab"),
    )
)

SOURCE_DIR = LAB_HOME / "data" / "source"
STAGING_DIR = LAB_HOME / "data" / "staging"
PROCESSED_DIR = LAB_HOME / "data" / "processed"
REJECTED_DIR = LAB_HOME / "data" / "rejected"
TARGET_DIR = LAB_HOME / "data" / "targets"
AUDIT_DIR = LAB_HOME / "data" / "audit"

ALLOWED_TABLES = {"Employee", "Department", "Customer", "Product"}
ALLOWED_ENVIRONMENTS = {"Development", "Testing", "Production"}
ALLOWED_STATUSES = {"ACTIVE", "INACTIVE", "PENDING"}
REQUIRED_COLUMNS = {"record_id", "name", "created_date", "status", "amount"}


def _safe_run_id(run_id: str) -> str:
    return re.sub(r"[^A-Za-z0-9_.-]+", "_", run_id)


def _runtime_values(context: dict[str, Any]) -> dict[str, Any]:
    params = context["params"]

    values = {
        "table_name": str(params["table_name"]).strip(),
        "year": int(params["year"]),
        "month": str(params["month"]).strip(),
        "environment": str(params["environment"]).strip(),
        "confirm_production": bool(params["confirm_production"]),
    }

    if values["table_name"] not in ALLOWED_TABLES:
        raise ValueError("Unsupported table_name.")

    if values["environment"] not in ALLOWED_ENVIRONMENTS:
        raise ValueError("Unsupported environment.")

    if (
        values["environment"] == "Production"
        and not values["confirm_production"]
    ):
        raise ValueError(
            "Production was selected, but confirm_production is false."
        )

    return values


def show_parameters(**context) -> dict[str, Any]:
    values = _runtime_values(context)

    print("=" * 62)
    print("Received DAG Parameters")
    print("=" * 62)
    print(f"Table              : {values['table_name']}")
    print(f"Year               : {values['year']}")
    print(f"Month              : {values['month']}")
    print(f"Environment        : {values['environment']}")
    print(f"Confirm production : {values['confirm_production']}")
    print("=" * 62)

    return values


def extract_table(**context) -> dict[str, Any]:
    values = context["ti"].xcom_pull(task_ids="show_parameters")
    run_folder = _safe_run_id(str(context["run_id"]))

    source_file = SOURCE_DIR / (
        f"{values['table_name']}_{values['year']}_{values['month']}.csv"
    )
    staging_folder = STAGING_DIR / run_folder
    staging_folder.mkdir(parents=True, exist_ok=True)
    staged_file = staging_folder / source_file.name

    if not source_file.exists():
        raise FileNotFoundError(
            "Source file not found. Expected: "
            f"{source_file}"
        )

    shutil.copy2(source_file, staged_file)

    print(f"Source file : {source_file}")
    print(f"Staged file : {staged_file}")

    return {
        **values,
        "run_folder": run_folder,
        "source_file": str(source_file),
        "staged_file": str(staged_file),
    }


def validate_table(**context) -> dict[str, Any]:
    metadata = context["ti"].xcom_pull(task_ids="extract_table")
    staged_file = Path(metadata["staged_file"])
    run_folder = metadata["run_folder"]

    processed_folder = PROCESSED_DIR / run_folder
    rejected_folder = REJECTED_DIR / run_folder
    processed_folder.mkdir(parents=True, exist_ok=True)
    rejected_folder.mkdir(parents=True, exist_ok=True)

    clean_file = processed_folder / "valid_records.csv"
    rejected_file = rejected_folder / "rejected_records.csv"

    valid_rows: list[dict[str, Any]] = []
    rejected_rows: list[dict[str, Any]] = []

    with staged_file.open("r", encoding="utf-8", newline="") as source:
        reader = csv.DictReader(source)
        columns = set(reader.fieldnames or [])

        missing_columns = REQUIRED_COLUMNS - columns
        if missing_columns:
            raise ValueError(
                "Missing required columns: "
                + ", ".join(sorted(missing_columns))
            )

        for line_number, raw in enumerate(reader, start=2):
            row = {key: (value or "").strip() for key, value in raw.items()}
            reasons: list[str] = []

            if not row["record_id"]:
                reasons.append("MISSING_RECORD_ID")

            if not row["name"]:
                reasons.append("MISSING_NAME")

            try:
                pendulum.parse(row["created_date"], strict=True)
            except Exception:
                reasons.append("INVALID_CREATED_DATE")

            status = row["status"].upper()
            if status not in ALLOWED_STATUSES:
                reasons.append("INVALID_STATUS")

            try:
                amount = float(row["amount"])
                if amount < 0:
                    reasons.append("NEGATIVE_AMOUNT")
            except ValueError:
                amount = 0.0
                reasons.append("INVALID_AMOUNT")

            transformed = {
                "record_id": row["record_id"].upper(),
                "name": row["name"].title(),
                "created_date": row["created_date"],
                "status": status,
                "amount": round(amount, 2),
                "source_table": metadata["table_name"],
                "processing_year": metadata["year"],
                "processing_month": metadata["month"],
                "target_environment": metadata["environment"],
                "source_line_number": line_number,
            }

            if reasons:
                transformed["rejection_reason"] = "|".join(reasons)
                rejected_rows.append(transformed)
            else:
                valid_rows.append(transformed)

    clean_fields = [
        "record_id",
        "name",
        "created_date",
        "status",
        "amount",
        "source_table",
        "processing_year",
        "processing_month",
        "target_environment",
        "source_line_number",
    ]

    with clean_file.open("w", encoding="utf-8", newline="") as target:
        writer = csv.DictWriter(target, fieldnames=clean_fields)
        writer.writeheader()
        writer.writerows(valid_rows)

    with rejected_file.open("w", encoding="utf-8", newline="") as target:
        writer = csv.DictWriter(
            target,
            fieldnames=clean_fields + ["rejection_reason"],
        )
        writer.writeheader()
        writer.writerows(rejected_rows)

    if not valid_rows:
        raise ValueError("No valid records remained after validation.")

    print(f"Valid rows    : {len(valid_rows)}")
    print(f"Rejected rows : {len(rejected_rows)}")
    print(f"Clean file    : {clean_file}")
    print(f"Rejected file : {rejected_file}")

    return {
        **metadata,
        "clean_file": str(clean_file),
        "rejected_file": str(rejected_file),
        "valid_count": len(valid_rows),
        "rejected_count": len(rejected_rows),
    }


def load_table(**context) -> dict[str, Any]:
    metadata = context["ti"].xcom_pull(task_ids="validate_table")
    clean_file = Path(metadata["clean_file"])

    target_folder = (
        TARGET_DIR
        / metadata["environment"]
        / metadata["table_name"]
        / str(metadata["year"])
        / metadata["month"]
    )
    target_folder.mkdir(parents=True, exist_ok=True)

    target_file = target_folder / "loaded_records.csv"
    shutil.copy2(clean_file, target_file)

    print(
        f"Loaded {metadata['valid_count']} records from "
        f"{metadata['table_name']} into {metadata['environment']}."
    )
    print(f"Target file: {target_file}")

    return {
        **metadata,
        "target_file": str(target_file),
    }


def create_audit_summary(**context) -> str:
    metadata = context["ti"].xcom_pull(task_ids="load_table")
    run_folder = metadata["run_folder"]

    AUDIT_DIR.mkdir(parents=True, exist_ok=True)
    audit_file = AUDIT_DIR / f"audit_{run_folder}.json"

    audit = {
        "dag_id": context["dag"].dag_id,
        "run_id": str(context["run_id"]),
        "table_name": metadata["table_name"],
        "year": metadata["year"],
        "month": metadata["month"],
        "environment": metadata["environment"],
        "valid_count": metadata["valid_count"],
        "rejected_count": metadata["rejected_count"],
        "source_file": metadata["source_file"],
        "target_file": metadata["target_file"],
        "status": "SUCCESS",
    }

    audit_file.write_text(
        json.dumps(audit, indent=2),
        encoding="utf-8",
    )

    print(json.dumps(audit, indent=2))
    return str(audit_file)


with DAG(
    dag_id="parameterized_table_processing",
    description="Reusable ETL workflow driven by validated runtime parameters.",
    schedule=None,
    start_date=pendulum.datetime(2026, 1, 1, tz="Asia/Kolkata"),
    catchup=False,
    max_active_runs=2,
    params={
        "table_name": Param(
            default="Employee",
            type="string",
            enum=["Employee", "Department", "Customer", "Product"],
            title="Source table",
            description="Select the business table to process.",
            section="Source",
        ),
        "year": Param(
            default=2026,
            type="integer",
            minimum=2020,
            maximum=2030,
            title="Processing year",
            section="Processing period",
        ),
        "month": Param(
            default="August",
            type="string",
            enum=[
                "January", "February", "March", "April",
                "May", "June", "July", "August",
                "September", "October", "November", "December",
            ],
            title="Processing month",
            section="Processing period",
        ),
        "environment": Param(
            default="Development",
            type="string",
            enum=["Development", "Testing", "Production"],
            title="Target environment",
            section="Target",
        ),
        "confirm_production": Param(
            default=False,
            type="boolean",
            title="Confirm Production load",
            description=(
                "Select true only when Target environment is Production."
            ),
            section="Target",
        ),
    },
    tags=["parameterized", "etl", "training"],
) as dag:
    start = EmptyOperator(task_id="start")

    show_parameters_task = PythonOperator(
        task_id="show_parameters",
        python_callable=show_parameters,
    )

    extract_task = PythonOperator(
        task_id="extract_table",
        python_callable=extract_table,
    )

    validate_task = PythonOperator(
        task_id="validate_table",
        python_callable=validate_table,
    )

    load_task = PythonOperator(
        task_id="load_table",
        python_callable=load_table,
    )

    audit_task = PythonOperator(
        task_id="create_audit_summary",
        python_callable=create_audit_summary,
    )

    finish = EmptyOperator(task_id="finish")

    (
        start
        >> show_parameters_task
        >> extract_task
        >> validate_task
        >> load_task
        >> audit_task
        >> finish
    )

14. Validation and Security

  • Use enum allowlists for identifiers.
  • Do not concatenate unrestricted table names into SQL.
  • Do not pass passwords or tokens as Params.
  • Use Connections or a secrets backend.
  • Require confirmation for Production.
  • Audit parameter values and outputs.
  • Keep loads idempotent.

16. Best Practices and Edge Cases

Best practices

  • Consistent names
  • Typed Params
  • Enums for identifiers
  • Meaningful defaults
  • Audit runs
  • Idempotent loads

Edge cases

  • Missing source combination
  • No valid rows
  • Overlapping writes
  • Wrong environment
  • Unexpected scheduled defaults
  • Changed parameter schema

17. Interview Q&A

Why use a parameterized DAG?
Reuse one workflow with different values and reduce duplicate code.
params versus Param?
params is the DAG mapping; Param defines validation and form metadata for a value.
How does PythonOperator access values?
context["params"]["table_name"].
Params versus tags?
Params are runtime inputs; tags group DAGs in the UI.
Params versus Variables?
Params are per-run; Variables persist across runs.
Why enum for table name?
It provides an allowlist, prevents typing errors and reduces injection risk.
How do scheduled runs get Params?
They use valid DAG defaults.
Why not pass secrets?
Run config and logs can expose them; use Connections or a secrets backend.
What is idempotency?
The same parameter combination can be rerun without corrupting the result.
Can one run process many tables?
Yes, with a list Param and dynamic task mapping as an advanced design.
